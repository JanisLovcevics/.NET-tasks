int[] sk_values = [-1, 0, 1, 9, 10];
bool[,] matrix = new bool[sk_values.Length, 5];

int[] current_answer = [0,0];

Console.Clear();

Console.WriteLine("skaitlis\tskaitlis > 0\tskaitlis < 10\tskaitlis > 0 && skailtis < 10\t!(skaitlis > 0 && skailtis < 10)\tskaitlis > 0 || skailtis < 10");
for (int i = 0; i < matrix.GetLength(0); i++)
{
    Console.Write("  " + sk_values[i] + "\t\t   ");

    for (int j = 0; j < matrix.GetLength(1); j++)
    {
        if (i == current_answer[0] && j == current_answer[1])
        {
            int cursor = Console.CursorLeft;
            matrix[i, j] = Convert.ToBoolean(Console.ReadLine());
            Console.CursorTop -= 1;
            Console.CursorLeft = cursor += Convert.ToString(matrix[i, j]).Length;
            current_answer[1]++;
        }
        else
        {
            Console.Write(matrix[i, j]);
        }
        Console.Write("\t\t");
        if (j > 1)
        {
            Console.Write("\t\t");
        }
        if (j > 2)
        {
            Console.Write("\t");
        }
    }

    Console.WriteLine();
    current_answer[0]++;
}
